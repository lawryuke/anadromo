using System;
using System.Net;
using System.Net.Sockets;
using System.Text;
using System.Threading;
using UnityEngine;

[Serializable]
public class SwimAction
{
    public string action;
    public float speed;
}

/// <summary>
/// Recibe comandos UDP del servidor Python (pose_swim_server.py)
/// y mueve este GameObject en consecuencia.
/// Formato esperado: {"action":"forward|idle","speed":float}
/// Nota: El giro lo controla el headset VR (no se reciben turn_left/turn_right).
/// </summary>
public class FishSwimController : MonoBehaviour
{
    public int listenPort = 5065;
    public float forwardMultiplier = 2f;

    [Header("Head-Joystick Rotation")]
    public Transform headCamera; // Asignar la Main Camera del XR Origin
    public float turnSpeed = 60f; // Grados por segundo al girar a tope
    public float turnDeadzone = 15f; // Grados de tolerancia antes de empezar a girar

    private UdpClient udpClient;
    private Thread receiveThread;
    private volatile string latestJson = null;
    private readonly object lockObj = new object();

    void Start()
    {
        udpClient = new UdpClient(listenPort);
        receiveThread = new Thread(ReceiveLoop) { IsBackground = true };
        receiveThread.Start();
    }

    private void ReceiveLoop()
    {
        IPEndPoint remoteEP = new IPEndPoint(IPAddress.Any, listenPort);
        while (true)
        {
            try
            {
                byte[] data = udpClient.Receive(ref remoteEP);
                string json = Encoding.UTF8.GetString(data);
                lock (lockObj) { latestJson = json; }
            }
            catch (Exception)
            {
                break; // socket cerrado al salir
            }
        }
    }

    void Update()
    {
        // 1. Manejar la rotación continua (Joystick de Cabeza)
        if (headCamera != null)
        {
            // Obtener el Yaw local de la cabeza respecto al cuerpo
            float localYaw = headCamera.localEulerAngles.y;
            if (localYaw > 180f) localYaw -= 360f; // Normalizar a -180..180

            if (localYaw > turnDeadzone)
            {
                // Girar derecha
                float excess = localYaw - turnDeadzone;
                float speedRatio = Mathf.Clamp01(excess / 30f); // Máxima velocidad a +30 grados del umbral
                transform.Rotate(0, speedRatio * turnSpeed * Time.deltaTime, 0);
            }
            else if (localYaw < -turnDeadzone)
            {
                // Girar izquierda
                float excess = Mathf.Abs(localYaw) - turnDeadzone;
                float speedRatio = Mathf.Clamp01(excess / 30f);
                transform.Rotate(0, -speedRatio * turnSpeed * Time.deltaTime, 0);
            }
        }

        // 2. Manejar el movimiento frontal desde UDP
        string json;
        lock (lockObj) { json = latestJson; }
        if (string.IsNullOrEmpty(json)) return;

        SwimAction action = JsonUtility.FromJson<SwimAction>(json);
        if (action == null) return;

        switch (action.action)
        {
            case "forward":
                transform.Translate(Vector3.forward * action.speed * forwardMultiplier * Time.deltaTime);
                break;
            case "idle":
            default:
                break;
        }
    }

    void OnApplicationQuit()
    {
        try { receiveThread?.Abort(); } catch { /* ignore */ }
        udpClient?.Close();
    }
}
